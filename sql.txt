CREATE PROCEDURE delete_by_ids_json (
    IN p_table_name VARCHAR(64),
    IN p_ids_json JSON
)
BEGIN

    CREATE TEMPORARY TABLE IF NOT EXISTS tmp_delete_ids (
        id BIGINT PRIMARY KEY
    ) ENGINE=InnoDB;

    TRUNCATE TABLE tmp_delete_ids;

    /* explode JSON array into rows */
    INSERT IGNORE INTO tmp_delete_ids (id)
    SELECT jt.id
    FROM JSON_TABLE(
        p_ids_json,
        '$[*]' COLUMNS (
            id BIGINT PATH '$'
        )
    ) AS jt;

    /* dynamic delete */
    SET @sql = CONCAT(
        'DELETE t FROM ',
        p_table_name,
        ' t JOIN tmp_delete_ids d ON t.id = d.id'
    );

    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;

END$$

DELIMITER ;

CREATE PROCEDURE insert_rows_json2 (
    IN p_table_name VARCHAR(64),
    IN p_rows JSON
)
BEGIN

    /* staging table */
    CREATE TEMPORARY TABLE IF NOT EXISTS tmp_insert_rows (
        id BIGINT,
        p CHAR(100),
        PRIMARY KEY (id)
    ) ENGINE=InnoDB;

    TRUNCATE TABLE tmp_insert_rows;

    /* explode JSON rows */
    INSERT INTO tmp_insert_rows (id, p)
    SELECT
        jt.id,
        jt.p
    FROM JSON_TABLE(
        p_rows,
        '$[*]' COLUMNS (
            id BIGINT PATH '$.id',
            p INT PATH '$.p'
        )
    ) AS jt;

    /* ordered bulk insert */
    SET @sql = CONCAT(
        'INSERT INTO ', p_table_name,
        ' (id, payload)
         SELECT id, p
         FROM tmp_insert_rows
         ORDER BY id'
    );

    PREPARE stmt FROM @sql;
    EXECUTE stmt;
    DEALLOCATE PREPARE stmt;

END$$

DELIMITER ;
